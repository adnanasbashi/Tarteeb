// PWA Prompt Management let deferredPrompt; window.addEventListener('beforeinstallprompt', (e) => { // Prevent standard banner prompt e.preventDefault(); // Stash the event so it can be triggered on user action deferredPrompt = e; // Reveal the Install action button const installBtn = document.getElementById('installBtn'); if (installBtn) installBtn.style.display = 'inline-block'; }); async function installApp() { if (!deferredPrompt) { toast('App is already installed or prompt unavailable.'); return; } // Show prompt to user deferredPrompt.prompt(); const { outcome } = await deferredPrompt.userChoice; if (outcome === 'accepted') { toast('Thank you for installing Tarteeb!'); } deferredPrompt = null; document.getElementById('installBtn').style.display = 'none'; } window.addEventListener('appinstalled', () => { deferredPrompt = null; document.getElementById('installBtn').style.display = 'none'; });
